${item.namn}

${metaText}

${getButtonHtml(item.pdf, 'Karta (PDF)', 'btn-pdf')} ${getButtonHtml(item.winsplits, 'Split Times', 'btn-winsplits')} ${getButtonHtml(item.livelox, 'Route', 'btn-livelox')}
${item.referat}
